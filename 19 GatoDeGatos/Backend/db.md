CREATE DATABASE IF NOT EXISTS gato_de_gatos CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE gato_de_gatos;

CREATE TABLE IF NOT EXISTS jugadores (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(80) NOT NULL,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS partidas (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  jugador_x_id INT UNSIGNED NOT NULL,
  jugador_o_id INT UNSIGNED NOT NULL,
  ganador ENUM('X','O','EMPATE') DEFAULT NULL,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  terminado_en TIMESTAMP NULL,
  FOREIGN KEY (jugador_x_id) REFERENCES jugadores(id),
  FOREIGN KEY (jugador_o_id) REFERENCES jugadores(id)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS puntajes (
  usuario_id INT UNSIGNED PRIMARY KEY,
  partidas_jugadas INT UNSIGNED DEFAULT 0,
  partidas_ganadas INT UNSIGNED DEFAULT 0,
  partidas_perdidas INT UNSIGNED DEFAULT 0,
  partidas_empatadas INT UNSIGNED DEFAULT 0,
  puntaje INT DEFAULT 0,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (usuario_id) REFERENCES jugadores(id)
) ENGINE=InnoDB;
